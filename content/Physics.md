$$
\text{Given that }\\ 
\sum F_{1x}=0\\
\sum F_{1y}=T-m_{1}g=m_{1}a \\
\text{and } \\
\sum F_{2x}=F\cos \theta -T-f=m_{2}a\\
\sum F_{2y}=F\sin \theta+n-m_{2}g \\
\dots \\ \\
F\cos \theta-T-f=m_{2}a\\
\implies F\cos \theta-T-\mu n=m_{2}a \\ \\
F\sin \theta +n-m_{2}g=0 \\
\implies n=m_{2}g-F\sin \theta \\ \\
\text{Performing addition of functions on }\sum F_{1y} \text{ and }\sum F_{2x} \text{ gives us}\\ \\
\text{Let } \sum F_{1y}=T-m_{1}g=m_{2}a \text{ and } \sum F_{2x}=F\cos \theta-T-\mu (m_{2}g-F\sin \theta)\\ \\ \text{Observe that}\\
\\
T-m_{1}g+(F\cos \theta-T-\mu(m_{2}g-F\sin \theta))=m_{1}a+m_{2}a \\
\implies F\cos \theta-m_{1}g-\mu(m_{2}g-F\sin \theta)=a(m_{1}+m_{2})\\
\implies \frac{F\cos \theta-m_{1}g-\mu(m_{2}g-F\sin \theta)}{m_{1}+m_{2}}=a \\ \\
\text{To find }T\text{ we can substitute }a\text{ to }\sum F_{1y} \\
\text{Observe that }\\ \\
T-m_{1}g=m_{1}\left(\frac{F\cos \theta-m_{1}g-\mu(m_{2}g-F\sin \theta)}{m_{1}+m_{2}}\right) \\
\implies T=m_{1}\left(\frac{F\cos \theta-m_{1}g-\mu(m_{2}g-F\sin \theta)}{m_{1}+m_{2}}\right) +m_{1}g \\
\implies T=\frac{m_{1}F\cos \theta-m_{1}^{2}g-m_{1} \mu(m_{2}g-F\sin \theta)}{m_{1}+m_{2}} +m_{1}g\\
\implies T=\frac{m_{1}F\cos \theta-m_{1}^{2}g-m_{1} \mu(m_{2}g-F\sin \theta)+m_{1}g(m_{1}+m_{2})}{m_{1}+m_{2}}\\
\implies T=m_{1}\left(\frac{F\cos \theta-m_{1}g-\mu(m_{2}g-F\sin \theta)+g(m_{1}+m_{2})}{m_{1}+m_{2}}\right)\\
\implies T=m_{1}\left(\frac{F\cos \theta-m_{1}g-\mu m_{2}g-\mu F\sin \theta+m_{1}g+m_{2}g}{m_{1}+m_{2}}\right)\\
\implies T=m_{1}\left(\frac{F\cos \theta-\mu m_{2}g-\mu F\sin \theta+m_{2}g}{m_{1}+m_{2}}\right)\\
\implies T=m_{1}\left(\frac{F\cos \theta-\mu F\sin \theta-\mu m_{2}g+m_{2}g}{m_{1}+m_{2}}\right)\\
\implies T=m_{1}\left[\frac{F(\cos \theta-\mu \sin \theta)+m_{2}g(-\mu+1)}{m_{1}+m_{2}}\right]\\
\Box

Example on Block Drop Distance

$$
\text{Given that } \\
\sum F_{Ax}=T-f=m_{A}a\\
\sum F_{Ay}=W_{A}-n=0\\
\text{and } \\
\sum F_{Bx}=0\\
\sum F_{By}=W_{B}-T=m_{B}a\\ \\
\text{If we let } \\
f_{k}=\mu_{k}n \text{ and } W=mg\\ \\
\sum F_{Ax}=T-\mu n=m_{A}a\\
\sum F_{Ay}=m_{A}g-n=0\\
\sum F_{Bx}=0\\
\sum F_{By}=m_{B}g-T=m_{B}a\\
\\
\text{Solving for } n \text{ give us}\\
m_{A}g-n=0\\
\implies m_{A}g-n+n=0+n\\
\implies m_{A}g=n\\ \\
\text{Letting } n=m_{A}g \text{ gives us the following equations:}\\
\sum F_{Ax}=T-\mu m_{A}g=m_{A}a\\
\sum F_{By}=m_{B}g-T=m_{B}a\\
\\
\text{Performing addition of equations on } \sum F_{Ax} \text{ and }\sum F_{By}\text{ gives us:}\\
T-T+m_{B}g-\mu m_{A}g=m_{A}a+m_{B}a\\
\implies m_{B}g-\mu m_{A}g=a(m_{A}+m_{B})\\
\\
\text{Let } m_{A}=25\text{kg},\text{ }m_{B}=15\text{kg},\text{ }g=9.8\text{m/s},\text{ and }\mu=0.2\\
m_{B}g-\mu m_{A}g=a(m_{A}+m_{B})\\
\implies (15\text{kg})\left(9.8\frac{m}{s^2}\right)-0.2 (25\text{kg})\left(9.8\frac{m}{s^2}\right)=a(25\text{kg}+15\text{kg})\\
\implies 147\text{N}-49\text{N}=a(40\text{kg})\\
\implies \frac{98\text{N}}{40\text{kg}}=a\left( \frac{40\text{kg}}{40\text{kg}} \right)\\
\implies 2.45\frac{m}{s^{2}}=a\\ \\
\text{To solve for the drop distance of Block B after 3.0s we are given that:}\\
y=y_{0}+v_{0y}t+\frac{1}{2}at^{2} \\ \\
\text{Let } y_{0}=0, v_{0}=0,t=3.0s,a=2.45 \frac{m}{s^{2}}\\
y=0+0(3.0s)+\frac{1}{2}2.45 \frac{m}{s^{2}}(3.0s)^{2}\\
\implies y=\frac{1}{2}2.45 \frac{m}{s^{2}}(9.0s^{2})\\
\implies y=\frac{1}{2}22.05m\\
\implies y=11.025m\\
\therefore \text{Block B dropped }11.025m\text{ after the system is released.} \Box
